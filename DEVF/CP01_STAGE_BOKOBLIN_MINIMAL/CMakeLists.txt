cmake_minimum_required(VERSION 3.25)
project(DEVFStageBokoblinMinimal LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 MSVC required")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_stage_bokoblin_minimal FEATURES game SOURCES src/mod.cpp MOD_JSON mod.json)
target_compile_options(devf_stage_bokoblin_minimal PRIVATE /utf-8 /EHsc /Zc:__cplusplus /GL-)
target_link_options(devf_stage_bokoblin_minimal PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_stage_bokoblin_minimal.map")
add_executable(stage_bokoblin_lab EXCLUDE_FROM_ALL tests/stage_lab.cpp src/mod.cpp)
target_link_libraries(stage_bokoblin_lab PRIVATE dusklight_mod_api dusklight_game_abi_headers)
target_compile_features(stage_bokoblin_lab PRIVATE cxx_std_20)
target_compile_options(stage_bokoblin_lab PRIVATE /FIglobal.h /utf-8 /EHsc /Zc:__cplusplus /GL-)
