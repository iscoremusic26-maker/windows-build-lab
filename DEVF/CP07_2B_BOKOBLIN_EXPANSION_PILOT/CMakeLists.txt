cmake_minimum_required(VERSION 3.25)
project(DEVFEncounterExpansion LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 MSVC only")
endif()
string(LENGTH "${DEVF_SOURCE_SHA}" source_len)
if(NOT source_len EQUAL 40 OR NOT DEVF_SOURCE_SHA MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "Pin source SHA")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_native_api_provider FEATURES game SOURCES provider/src/mod.cpp provider/src/native_capture.cpp MOD_JSON provider/mod.json)
add_mod(bokoblin_expansion SOURCES expansion/src/mod.cpp MOD_JSON expansion/mod.json)
foreach(target IN ITEMS devf_native_api_provider bokoblin_expansion)
 target_compile_options(${target} PRIVATE /utf-8 /EHsc /Zc:__cplusplus /GL-)
 target_compile_definitions(${target} PRIVATE DEVF_SOURCE_SHA="${DEVF_SOURCE_SHA}")
endforeach()
add_executable(native_config_lab provider/tests/config_lab.cpp)
add_executable(editor_lab provider/tests/editor_lab.cpp)
add_executable(api_lab tests/api_lab.cpp)
add_executable(encounter_lab tests/encounter_lab.cpp)
add_executable(encounter_validate tests/encounter_validate.cpp)
add_executable(expansion_lab tests/expansion_lab.cpp)
foreach(target IN ITEMS native_config_lab editor_lab api_lab encounter_lab encounter_validate expansion_lab)
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_options(${target} PRIVATE /utf-8 /EHsc /GL-)
 target_include_directories(${target} PRIVATE "${DUSK_SOURCE}/sdk/include")
endforeach()
add_library(api_c_abi OBJECT tests/api_c_abi.c)
target_compile_features(api_c_abi PRIVATE c_std_11)
target_include_directories(api_c_abi PRIVATE "${DUSK_SOURCE}/sdk/include")

add_library(encounter_api_c_abi OBJECT tests/encounter_api_c_abi.c)
target_compile_features(encounter_api_c_abi PRIVATE c_std_11)
target_include_directories(encounter_api_c_abi PRIVATE "${DUSK_SOURCE}/sdk/include")
