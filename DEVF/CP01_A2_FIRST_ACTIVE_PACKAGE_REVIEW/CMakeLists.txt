cmake_minimum_required(VERSION 3.25)
project(DEVFFirstActivePackage LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64/MSVC required; no Linux simulation")
endif()
if(NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_GAME_EXE}")
 message(FATAL_ERROR "Exact SDK and CP00 import library required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(PILOT "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_FIRST_PILOT_SAFETY_AND_TRACE")
set(ACTIVE_TRACE "${CMAKE_CURRENT_BINARY_DIR}/active_trace")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ci/prepare_trace.py" --pilot "${PILOT}" --output "${ACTIVE_TRACE}" RESULT_VARIABLE TRACE_RESULT)
if(NOT TRACE_RESULT EQUAL 0)
 message(FATAL_ERROR "Trace adaptation failed")
endif()
set(OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/overlay")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PILOT}/ci/build_overlay.py" --project "${PILOT}" --output "${OVERLAY}" RESULT_VARIABLE OVERLAY_RESULT)
if(NOT OVERLAY_RESULT EQUAL 0)
 message(FATAL_ERROR "Reviewed source overlay failed")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
set(G2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G2_READONLY_PACKAGE")
set(A2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_DISARMED_CONTROLLER")
set(G4 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G4B_LOCAL_READONLY_OBSERVER")
set(OLD "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_FIRST_NATIVE_REQUEST_OFFLINE")
# Only the read-only bridge receives unchanged historical guards.
add_library(pilot_readonly_bridge STATIC EXCLUDE_FROM_ALL "${PILOT}/src/readonly_bridge.cpp" "${PILOT}/src/census_values.cpp" "${G4}/src/capture.cpp" "${G4}/src/local_queries.cpp" "${OVERLAY}/src/census_reader.cpp" "${G2}/src/core.cpp" "${A2}/src/controller_core.cpp" "${A2}/src/room_owner_resolver.cpp")
target_link_libraries(pilot_readonly_bridge PRIVATE dusklight_mod_api dusklight_game_abi_headers)
target_include_directories(pilot_readonly_bridge PRIVATE "${OVERLAY}/src" "${PILOT}/src" "${G4}/src")
target_compile_definitions(pilot_readonly_bridge PRIVATE DEVF_FIRST_OFFLINE=1 DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1 DUSK_MOD_FEATURE_GAME=1)
target_compile_features(pilot_readonly_bridge PRIVATE cxx_std_20)
target_compile_options(pilot_readonly_bridge PRIVATE /FIglobal.h /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
# Only official add_mod packaging, never BUNDLE/install/host build.
add_mod(devf_first_eoc_active FEATURES game MOD_JSON mod.json SOURCES "${OVERLAY}/src/experiment_core.cpp" "${OVERLAY}/src/native_backend.cpp" src/active_admission.cpp src/active_mod.cpp "${PILOT}/src/policy.cpp" "${ACTIVE_TRACE}/trace.cpp")
target_link_libraries(devf_first_eoc_active PRIVATE dusklight_mod_feature_game pilot_readonly_bridge "${DUSK_GAME_EXE}")
target_include_directories(devf_first_eoc_active PRIVATE "${ACTIVE_TRACE}" "${OVERLAY}/src" src "${PILOT}/src")
target_compile_definitions(devf_first_eoc_active PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(devf_first_eoc_active PRIVATE cxx_std_20)
target_compile_options(devf_first_eoc_active PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
target_link_options(devf_first_eoc_active PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_first_eoc_active.map" /INCLUDE:devf_first_request_link_anchor)

# Fixtures are reused, not duplicated. Historical main is compiled but never invoked.
add_executable(active_provider_lab EXCLUDE_FROM_ALL tests/provider_lab.cpp "${OVERLAY}/src/experiment_core.cpp" src/active_admission.cpp "${PILOT}/src/policy.cpp" "${ACTIVE_TRACE}/trace.cpp" "${PILOT}/src/census_values.cpp")
target_link_libraries(active_provider_lab PRIVATE dusklight_mod_api)
target_include_directories(active_provider_lab PRIVATE "${ACTIVE_TRACE}" "${OVERLAY}/src" "${OVERLAY}/tests" src "${PILOT}/src")
target_compile_definitions(active_provider_lab PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(active_provider_lab PRIVATE cxx_std_20)
target_compile_options(active_provider_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus /GL-)
target_link_options(active_provider_lab PRIVATE /STACK:8388608)
