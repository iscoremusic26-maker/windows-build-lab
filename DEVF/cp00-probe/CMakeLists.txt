cmake_minimum_required(VERSION 3.25)
project(devf_cp00_probe LANGUAGES CXX)
message(STATUS "CP00 compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "CP00 Windows SDK: ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CP00 requires Windows x64 with MSVC or clang-cl and Windows SDK")
endif()
set(DUSK_DIR "" CACHE PATH "Exact verified Dusklight 2.0.3 source root")
if(NOT EXISTS "${DUSK_DIR}/sdk/CMakeLists.txt")
  message(FATAL_ERROR "Set DUSK_DIR to the verified 2.0.3 source root")
endif()
if(NOT DUSK_GAME_EXE OR NOT EXISTS "${DUSK_GAME_EXE}" OR NOT DUSK_GAME_EXE MATCHES "\\.lib$")
  message(FATAL_ERROR "Set DUSK_GAME_EXE to sdk/windows-amd64.lib from the verified Windows ZIP")
endif()
add_subdirectory("${DUSK_DIR}/sdk" dusk-sdk EXCLUDE_FROM_ALL)
set(DUSK_AURORA_DIR "" CACHE PATH "Aurora headers at the exact 2.0.3 gitlink revision")
if(NOT EXISTS "${DUSK_AURORA_DIR}/include/dolphin/types.h")
  message(FATAL_ERROR "Set DUSK_AURORA_DIR to Aurora 3227d76c60e1e782ca576610bce61c9e7744d8be")
endif()
target_include_directories(dusklight_game_abi_headers INTERFACE
  "${DUSK_AURORA_DIR}/include" "${DUSK_AURORA_DIR}/include/dolphin")
add_mod(devf_cp00_probe FEATURES game SOURCES src/mod.cpp MOD_JSON mod.json)
