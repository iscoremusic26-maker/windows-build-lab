cmake_minimum_required(VERSION 3.25)
project(devf_cp01_host_lifetime_lab LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Requires real Windows x64 MSVC/Windows SDK; no Linux SDK substitution")
endif()
message(STATUS "HOST_EXPERIMENT compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION} sdk=${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}")
if(NOT EXISTS "${DUSK_DIR}/sdk/include/mods/api.h")
 message(FATAL_ERROR "Exact copied/patched 2.0.3 source required")
endif()
add_library(unload_fixture MODULE ../CP01/tests/unload_fixture.cpp)
target_include_directories(unload_fixture PRIVATE "${DUSK_DIR}/sdk/include")
add_executable(host_lifetime_lab tests/host_lifetime_lab.cpp)
target_include_directories(host_lifetime_lab PRIVATE "${DUSK_DIR}/sdk/include" "${EXTRACTED_DIR}")
target_compile_features(host_lifetime_lab PRIVATE cxx_std_20)
target_compile_options(host_lifetime_lab PRIVATE /W4 /Zc:__cplusplus)
enable_testing()
add_test(NAME CP01_HOST_EXPERIMENT_incremental COMMAND host_lifetime_lab "$<TARGET_FILE:unload_fixture>")
