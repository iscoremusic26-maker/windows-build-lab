cmake_minimum_required(VERSION 3.25)
project(devf_cp01 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CP01 requires real Windows x64 MSVC and Windows SDK")
endif()
message(STATUS "CP01 compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "CP01 Windows SDK: ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}")
set(DUSK_DIR "" CACHE PATH "Verified exact 2.0.3 source root")
set(DUSK_AURORA_DIR "" CACHE PATH "Aurora 3227d76c60e1e782ca576610bce61c9e7744d8be headers")
if(NOT EXISTS "${DUSK_DIR}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_AURORA_DIR}/include/dolphin/types.h")
  message(FATAL_ERROR "Exact SDK and pinned Aurora headers required")
endif()
if(NOT DUSK_GAME_EXE OR NOT EXISTS "${DUSK_GAME_EXE}" OR NOT DUSK_GAME_EXE MATCHES "\\.lib$")
  message(FATAL_ERROR "DUSK_GAME_EXE must be sdk/windows-amd64.lib from verified Windows ZIP")
endif()
add_subdirectory("${DUSK_DIR}/sdk" dusk-sdk EXCLUDE_FROM_ALL)
target_include_directories(dusklight_game_abi_headers INTERFACE "${DUSK_AURORA_DIR}/include" "${DUSK_AURORA_DIR}/include/dolphin")
add_mod(devf_cp01 FEATURES game SOURCES src/mod.cpp src/controller.cpp src/rock.cpp
  MOD_JSON mod.json RES_DIR res)
target_compile_options(devf_cp01 PRIVATE /Zc:__cplusplus)
# Lab exercises the SAME controller and resource lifecycle with simulated service tables,
# without linking or running the game. It is a Windows executable, not proof of engine behavior.
add_executable(devf_cp01_lifecycle_lab tests/lifecycle_lab.cpp src/controller.cpp)
target_link_libraries(devf_cp01_lifecycle_lab PRIVATE dusklight_mod_api)
target_include_directories(devf_cp01_lifecycle_lab PRIVATE src)
target_compile_features(devf_cp01_lifecycle_lab PRIVATE cxx_std_20)
target_compile_options(devf_cp01_lifecycle_lab PRIVATE /Zc:__cplusplus)
enable_testing()
add_test(NAME CP01_affected_regression COMMAND devf_cp01_lifecycle_lab --affected)
execute_process(COMMAND python "${CMAKE_CURRENT_SOURCE_DIR}/ci/extract_host_audit.py"
  --source "${DUSK_DIR}" --out "${CMAKE_BINARY_DIR}/audit"
  RESULT_VARIABLE audit_result)
if(NOT audit_result EQUAL 0)
  message(FATAL_ERROR "Exact host cleanup extraction failed")
endif()
add_library(devf_cp01_unload_fixture MODULE tests/unload_fixture.cpp)
add_library(devf_cp01_pin_fixture MODULE tests/unload_fixture.cpp)
foreach(fixture devf_cp01_unload_fixture devf_cp01_pin_fixture)
  target_link_libraries(${fixture} PRIVATE dusklight_mod_api)
  target_compile_features(${fixture} PRIVATE cxx_std_20)
endforeach()
add_executable(devf_cp01_safety_lab tests/safety_lab.cpp src/controller.cpp)
target_link_libraries(devf_cp01_safety_lab PRIVATE dusklight_mod_api)
target_include_directories(devf_cp01_safety_lab PRIVATE src "${CMAKE_BINARY_DIR}/audit")
target_compile_features(devf_cp01_safety_lab PRIVATE cxx_std_20)
target_compile_options(devf_cp01_safety_lab PRIVATE /Zc:__cplusplus)
add_dependencies(devf_cp01_safety_lab devf_cp01_unload_fixture devf_cp01_pin_fixture)
add_test(NAME CP01_new_safety COMMAND devf_cp01_safety_lab
  "$<TARGET_FILE:devf_cp01_unload_fixture>" "$<TARGET_FILE:devf_cp01_pin_fixture>")
