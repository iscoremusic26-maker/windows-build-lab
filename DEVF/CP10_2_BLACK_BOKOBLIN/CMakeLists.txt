cmake_minimum_required(VERSION 3.25)
project(DEVFCP102Black LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 MSVC required")
endif()
string(LENGTH "${DEVF_SOURCE_SHA}" source_len)
if(NOT source_len EQUAL 40)
 message(FATAL_ERROR "Pin source SHA")
endif()
set(OLD "${CMAKE_CURRENT_SOURCE_DIR}/../CP07_2B_BOKOBLIN_EXPANSION_PILOT")
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
# Only the Expansion changes in CP10.2. The CP10 provider 0.10.0 .dusk is reused unchanged and is NOT rebuilt here.
add_mod(bokoblin_expansion FEATURES game SOURCES expansion/mod.cpp MOD_JSON expansion/mod.json)
target_compile_options(bokoblin_expansion PRIVATE /utf-8 /EHsc /Zc:__cplusplus /Zc:inline- /GL-)
target_compile_definitions(bokoblin_expansion PRIVATE DEVF_SOURCE_SHA="${DEVF_SOURCE_SHA}")
add_executable(map_lab tests/map_lab.cpp)
target_compile_features(map_lab PRIVATE cxx_std_20)
target_include_directories(map_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
target_compile_options(map_lab PRIVATE /utf-8 /EHsc)
add_executable(black_lab tests/black_lab.cpp)
target_compile_features(black_lab PRIVATE cxx_std_20)
target_compile_options(black_lab PRIVATE /utf-8 /EHsc /GL-)
add_library(api_c_abi OBJECT "${OLD}/tests/api_c_abi.c" "${OLD}/tests/encounter_api_c_abi.c")
target_include_directories(api_c_abi PRIVATE "${DUSK_SOURCE}/sdk/include")
target_compile_features(api_c_abi PRIVATE c_std_11)
add_executable(legacy_lab ../CP08_MAP_EDITOR_BOKOBLIN_EXPANSION/tests/map_lab.cpp)
target_compile_features(legacy_lab PRIVATE cxx_std_20)
target_include_directories(legacy_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
target_compile_options(legacy_lab PRIVATE /utf-8 /EHsc)
