cmake_minimum_required(VERSION 3.25)
project(DEVFNativeSpawnQA LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Native QA requires Windows AMD64 MSVC")
endif()
string(LENGTH "${DEVF_SOURCE_SHA}" source_len)
if(NOT source_len EQUAL 40 OR NOT DEVF_SOURCE_SHA MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "Pin full source commit")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
foreach(mode IN ITEMS north dual)
 add_mod(devf_native_qa_${mode} SOURCES src/mod.cpp MOD_JSON manifests/${mode}.json)
 target_include_directories(devf_native_qa_${mode} PRIVATE src)
 target_compile_options(devf_native_qa_${mode} PRIVATE /utf-8 /EHsc /Zc:__cplusplus /GL-)
 target_compile_definitions(devf_native_qa_${mode} PRIVATE DEVF_SOURCE_SHA="${DEVF_SOURCE_SHA}")
 if(mode STREQUAL "north")
  target_compile_definitions(devf_native_qa_${mode} PRIVATE DEVF_QA_MODE=1)
 else()
  target_compile_definitions(devf_native_qa_${mode} PRIVATE DEVF_QA_MODE=2)
 endif()
endforeach()
add_executable(native_registration_lab tests/registration_lab.cpp)
target_compile_features(native_registration_lab PRIVATE cxx_std_20)
target_compile_options(native_registration_lab PRIVATE /utf-8 /EHsc /GL-)
target_include_directories(native_registration_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
