cmake_minimum_required(VERSION 3.25)
project(DEVFMapRuntime LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 MSVC required")
endif()
string(LENGTH "${DEVF_SOURCE_SHA}" source_len)
if(NOT source_len EQUAL 40)
 message(FATAL_ERROR "Pin source SHA")
endif()
set(OLD "${CMAKE_CURRENT_SOURCE_DIR}/../CP07_2B_BOKOBLIN_EXPANSION_PILOT")
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_map_provider FEATURES game SOURCES provider/mod.cpp "${OLD}/provider/src/native_capture.cpp" MOD_JSON provider/mod.json)
add_mod(bokoblin_expansion SOURCES expansion/mod.cpp MOD_JSON expansion/mod.json)
foreach(t IN ITEMS devf_map_provider bokoblin_expansion)
 target_compile_options(${t} PRIVATE /utf-8 /EHsc /Zc:__cplusplus /GL-)
 target_compile_definitions(${t} PRIVATE DEVF_SOURCE_SHA="${DEVF_SOURCE_SHA}")
endforeach()
add_executable(map_lab tests/map_lab.cpp)
target_compile_features(map_lab PRIVATE cxx_std_20)
target_include_directories(map_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
target_compile_options(map_lab PRIVATE /utf-8 /EHsc)
add_library(api_c_abi OBJECT "${OLD}/tests/api_c_abi.c" "${OLD}/tests/encounter_api_c_abi.c")
target_include_directories(api_c_abi PRIVATE "${DUSK_SOURCE}/sdk/include")
target_compile_features(api_c_abi PRIVATE c_std_11)
