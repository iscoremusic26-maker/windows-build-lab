cmake_minimum_required(VERSION 3.25)
project(DEVFFirstNativeOffline LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Real Windows x64 MSVC validation required")
endif()
if(NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_GAME_EXE}")
 message(FATAL_ERROR "Pinned official SDK and CP00 import library required")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
set(G2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G2_READONLY_PACKAGE")
set(A2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_DISARMED_CONTROLLER")
set(G4 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G4B_LOCAL_READONLY_OBSERVER")
# Read-only historical code stays in its OWN compiled target; no guard weakened or applied to mutator units.
add_library(first_readonly_bridge STATIC EXCLUDE_FROM_ALL src/readonly_bridge.cpp "${G4}/src/capture.cpp" "${G4}/src/local_queries.cpp" "${G4}/src/extended_g2_reader.cpp" "${G2}/src/core.cpp" "${A2}/src/controller_core.cpp" "${A2}/src/room_owner_resolver.cpp")
target_link_libraries(first_readonly_bridge PRIVATE dusklight_mod_api dusklight_game_abi_headers)
target_compile_definitions(first_readonly_bridge PRIVATE DEVF_FIRST_OFFLINE=1 DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1 DUSK_MOD_FEATURE_GAME=1)
target_compile_features(first_readonly_bridge PRIVATE cxx_std_20)
target_compile_options(first_readonly_bridge PRIVATE /FIglobal.h /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
# No add_mod()/mod.json/package target: this experimental MODULE is never packaged or uploaded.
add_library(devf_first_native_experiment MODULE EXCLUDE_FROM_ALL src/experiment_core.cpp src/native_backend.cpp src/admission.cpp src/experimental_mod.cpp)
target_link_libraries(devf_first_native_experiment PRIVATE dusklight_mod_feature_game first_readonly_bridge "${DUSK_GAME_EXE}")
target_compile_definitions(devf_first_native_experiment PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(devf_first_native_experiment PRIVATE cxx_std_20)
target_compile_options(devf_first_native_experiment PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
target_link_options(devf_first_native_experiment PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_first_native_experiment.map" /INCLUDE:devf_first_request_link_anchor)
add_executable(first_request_lab EXCLUDE_FROM_ALL tests/request_lab.cpp src/experiment_core.cpp)
target_compile_definitions(first_request_lab PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(first_request_lab PRIVATE cxx_std_20)
target_compile_options(first_request_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus)
add_executable(first_equivalence_lab EXCLUDE_FROM_ALL tests/equivalence_lab.cpp src/experiment_core.cpp "${A2}/src/controller_core.cpp")
target_compile_definitions(first_equivalence_lab PRIVATE DEVF_FIRST_OFFLINE=1 DEVF_A2_DISARMED=1)
target_compile_features(first_equivalence_lab PRIVATE cxx_std_20)
target_compile_options(first_equivalence_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus)
add_executable(first_abi_lab EXCLUDE_FROM_ALL tests/abi_lab.cpp)
target_link_libraries(first_abi_lab PRIVATE dusklight_mod_api)
target_compile_features(first_abi_lab PRIVATE cxx_std_20)
target_compile_options(first_abi_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus)
