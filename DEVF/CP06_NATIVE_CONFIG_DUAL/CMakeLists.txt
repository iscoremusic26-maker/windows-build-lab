cmake_minimum_required(VERSION 3.25)
project(DEVFNativeConfigDual LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows AMD64 MSVC required")
endif()
string(LENGTH "${DEVF_SOURCE_SHA}" source_len)
if(NOT source_len EQUAL 40 OR NOT DEVF_SOURCE_SHA MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "Pin full source commit")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_cp06_native_config_dual SOURCES src/mod.cpp MOD_JSON mod.json)
target_include_directories(devf_cp06_native_config_dual PRIVATE src)
target_compile_options(devf_cp06_native_config_dual PRIVATE /utf-8 /EHsc /Zc:__cplusplus /GL-)
target_compile_definitions(devf_cp06_native_config_dual PRIVATE DEVF_SOURCE_SHA="${DEVF_SOURCE_SHA}")
add_executable(native_config_lab tests/config_lab.cpp)
target_compile_features(native_config_lab PRIVATE cxx_std_20)
target_compile_options(native_config_lab PRIVATE /utf-8 /EHsc /GL-)
target_include_directories(native_config_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
