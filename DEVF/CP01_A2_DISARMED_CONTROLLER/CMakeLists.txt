cmake_minimum_required(VERSION 3.25)
project(DEVFA2Disarmed LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required for target validation")
endif()
if(NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_GAME_EXE}")
  message(FATAL_ERROR "Pinned official SDK and exact private import library required")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
set(G2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G2_READONLY_PACKAGE")
add_mod(devf_a2_disarmed FEATURES game SOURCES src/disarmed_mod.cpp src/controller_core.cpp src/room_owner_resolver.cpp
  "${G2}/src/core.cpp" "${G2}/src/native_reader.cpp" MOD_JSON mod.json)
set_target_properties(devf_a2_disarmed_package PROPERTIES EXCLUDE_FROM_ALL TRUE)
target_compile_definitions(devf_a2_disarmed PRIVATE DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1)
target_compile_options(devf_a2_disarmed PRIVATE /Zc:__cplusplus /utf-8 /EHsc /FAs)
target_link_options(devf_a2_disarmed PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_a2_disarmed.map")
add_executable(a2_controller_lab tests/controller_lab.cpp src/controller_core.cpp)
target_compile_definitions(a2_controller_lab PRIVATE DEVF_A2_DISARMED=1 DEVF_A2_LAB_ONLY=1)
target_compile_features(a2_controller_lab PRIVATE cxx_std_20)
target_compile_options(a2_controller_lab PRIVATE /Zc:__cplusplus /utf-8 /EHsc)
# Offline-only link probe retains exported ABI addresses; never installed/executed.
add_library(a2_abi_link_probe SHARED tests/abi_link_probe.cpp)
target_link_libraries(a2_abi_link_probe PRIVATE dusklight_mod_feature_game "${DUSK_GAME_EXE}")
target_compile_features(a2_abi_link_probe PRIVATE cxx_std_20)
target_compile_options(a2_abi_link_probe PRIVATE /Zc:__cplusplus)
enable_testing()
add_test(NAME a2_controller_lab COMMAND a2_controller_lab)
add_executable(a2_adapter_lab tests/adapter_lab.cpp src/disarmed_mod.cpp src/controller_core.cpp
  "${DUSK_SOURCE}/sdk/src/game_feature.cpp")
target_link_libraries(a2_adapter_lab PRIVATE dusklight_mod_api)
target_compile_features(a2_adapter_lab PRIVATE cxx_std_20)
target_compile_options(a2_adapter_lab PRIVATE /Zc:__cplusplus /utf-8 /EHsc)
target_compile_definitions(a2_adapter_lab PRIVATE DEVF_A2_DISARMED=1)
add_test(NAME a2_adapter_lab COMMAND a2_adapter_lab)
