cmake_minimum_required(VERSION 3.25)
project(DEVFG4BReadOnly LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required for target validation")
endif()
if(NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_GAME_EXE}")
  message(FATAL_ERROR "Pinned official SDK and exact private import library required")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
set(G2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G2_READONLY_PACKAGE")
set(A2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_DISARMED_CONTROLLER")
add_mod(devf_g4b_readonly FEATURES game SOURCES src/mod.cpp src/capture.cpp src/local_queries.cpp src/extended_g2_reader.cpp
  "${G2}/src/core.cpp" "${A2}/src/controller_core.cpp" "${A2}/src/room_owner_resolver.cpp" MOD_JSON mod.json)
set_target_properties(devf_g4b_readonly_package PROPERTIES EXCLUDE_FROM_ALL TRUE)
target_compile_definitions(devf_g4b_readonly PRIVATE DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1)
target_compile_options(devf_g4b_readonly PRIVATE /Zc:__cplusplus /utf-8 /EHsc)
target_link_options(devf_g4b_readonly PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_g4b_readonly.map")
add_executable(g4b_capture_lab tests/capture_lab.cpp src/capture.cpp "${A2}/src/controller_core.cpp")
target_compile_definitions(g4b_capture_lab PRIVATE DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1)
target_compile_features(g4b_capture_lab PRIVATE cxx_std_20)
target_compile_options(g4b_capture_lab PRIVATE /Zc:__cplusplus /utf-8 /EHsc)
add_executable(g4b_adapter_lab tests/adapter_lab.cpp tests/fake_capture.cpp src/mod.cpp src/capture.cpp "${A2}/src/controller_core.cpp" "${DUSK_SOURCE}/sdk/src/game_feature.cpp")
target_link_libraries(g4b_adapter_lab PRIVATE dusklight_mod_api)
target_compile_definitions(g4b_adapter_lab PRIVATE DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1)
target_compile_features(g4b_adapter_lab PRIVATE cxx_std_20)
target_compile_options(g4b_adapter_lab PRIVATE /Zc:__cplusplus /utf-8 /EHsc)
enable_testing()
add_test(NAME g4b_capture_lab COMMAND g4b_capture_lab)
add_test(NAME g4b_adapter_lab COMMAND g4b_adapter_lab)
