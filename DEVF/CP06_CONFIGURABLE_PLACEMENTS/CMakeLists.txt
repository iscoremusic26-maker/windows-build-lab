cmake_minimum_required(VERSION 3.25)
project(DEVFCP06SingleSite LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 MSVC required")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
add_mod(devf_cp06_single_site FEATURES game SOURCES src/mod.cpp MOD_JSON mod.json)
target_include_directories(devf_cp06_single_site PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(devf_cp06_single_site PRIVATE /utf-8 /EHsc /Zc:__cplusplus /Zc:inline- /GL-)
target_link_options(devf_cp06_single_site PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_cp06_single_site.map")
add_executable(cp06_config_lab EXCLUDE_FROM_ALL tests/config_lab.cpp)
target_compile_features(cp06_config_lab PRIVATE cxx_std_20)
target_compile_options(cp06_config_lab PRIVATE /utf-8 /EHsc /GL-)
target_include_directories(cp06_config_lab PRIVATE "${DUSK_SOURCE}/sdk/include")
