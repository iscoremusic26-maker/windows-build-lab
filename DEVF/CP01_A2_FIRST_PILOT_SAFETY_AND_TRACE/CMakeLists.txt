cmake_minimum_required(VERSION 3.25)
project(DEVFFirstPilotOffline LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64/MSVC required; no Linux simulation")
endif()
if(NOT EXISTS "${DUSK_SOURCE}/sdk/CMakeLists.txt" OR NOT EXISTS "${DUSK_GAME_EXE}")
 message(FATAL_ERROR "Exact SDK and CP00 import library required")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/overlay")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ci/build_overlay.py" --project "${CMAKE_CURRENT_SOURCE_DIR}" --output "${OVERLAY}" RESULT_VARIABLE OVERLAY_RESULT)
if(NOT OVERLAY_RESULT EQUAL 0)
 message(FATAL_ERROR "Reviewed source overlay failed")
endif()
add_subdirectory("${DUSK_SOURCE}/sdk" sdk EXCLUDE_FROM_ALL)
set(G2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G2_READONLY_PACKAGE")
set(A2 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_DISARMED_CONTROLLER")
set(G4 "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_G4B_LOCAL_READONLY_OBSERVER")
set(OLD "${CMAKE_CURRENT_SOURCE_DIR}/../CP01_A2_FIRST_NATIVE_REQUEST_OFFLINE")
# Only the read-only bridge receives unchanged historical guards.
add_library(pilot_readonly_bridge STATIC EXCLUDE_FROM_ALL src/readonly_bridge.cpp src/census_values.cpp "${G4}/src/capture.cpp" "${G4}/src/local_queries.cpp" "${OVERLAY}/src/census_reader.cpp" "${G2}/src/core.cpp" "${A2}/src/controller_core.cpp" "${A2}/src/room_owner_resolver.cpp")
target_link_libraries(pilot_readonly_bridge PRIVATE dusklight_mod_api dusklight_game_abi_headers)
target_include_directories(pilot_readonly_bridge PRIVATE "${OVERLAY}/src" src "${G4}/src")
target_compile_definitions(pilot_readonly_bridge PRIVATE DEVF_FIRST_OFFLINE=1 DEVF_G4B_READ_ONLY=1 DEVF_A2_DISARMED=1 DEVF_READ_ONLY=1 DUSK_MOD_FEATURE_GAME=1)
target_compile_features(pilot_readonly_bridge PRIVATE cxx_std_20)
target_compile_options(pilot_readonly_bridge PRIVATE /FIglobal.h /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
# Explicit MODULE only: no SDK packaging helper, mod.json, install target or build ALL.
add_library(devf_first_pilot_experiment MODULE EXCLUDE_FROM_ALL "${OVERLAY}/src/experiment_core.cpp" "${OVERLAY}/src/native_backend.cpp" "${OVERLAY}/src/admission.cpp" src/experimental_mod.cpp src/policy.cpp src/trace.cpp)
target_link_libraries(devf_first_pilot_experiment PRIVATE dusklight_mod_feature_game pilot_readonly_bridge "${DUSK_GAME_EXE}")
target_include_directories(devf_first_pilot_experiment PRIVATE "${OVERLAY}/src" src)
target_compile_definitions(devf_first_pilot_experiment PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(devf_first_pilot_experiment PRIVATE cxx_std_20)
target_compile_options(devf_first_pilot_experiment PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /GL-)
target_link_options(devf_first_pilot_experiment PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/devf_first_pilot_experiment.map" /INCLUDE:devf_first_request_link_anchor)
add_executable(pilot_safety_trace_lab EXCLUDE_FROM_ALL tests/pilot_lab.cpp "${OVERLAY}/src/experiment_core.cpp" "${OVERLAY}/src/admission.cpp" src/policy.cpp src/trace.cpp src/census_values.cpp)
target_include_directories(pilot_safety_trace_lab PRIVATE "${OVERLAY}/src" "${OVERLAY}/tests" src)
target_compile_definitions(pilot_safety_trace_lab PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(pilot_safety_trace_lab PRIVATE cxx_std_20)
target_compile_options(pilot_safety_trace_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus)
target_link_options(pilot_safety_trace_lab PRIVATE /STACK:8388608) # offline fixed-array fixture stack only
add_executable(pilot_native_abi_lab EXCLUDE_FROM_ALL "${OLD}/tests/abi_lab.cpp")
target_link_libraries(pilot_native_abi_lab PRIVATE dusklight_mod_api)
target_compile_features(pilot_native_abi_lab PRIVATE cxx_std_20)
target_compile_options(pilot_native_abi_lab PRIVATE /EHsc /utf-8 /Zc:__cplusplus)

add_executable(pilot_baseline_reentry_reproducer EXCLUDE_FROM_ALL tests/preflight_reentry_baseline.cpp "${OLD}/src/experiment_core.cpp")
target_compile_definitions(pilot_baseline_reentry_reproducer PRIVATE DEVF_FIRST_OFFLINE=1)
target_compile_features(pilot_baseline_reentry_reproducer PRIVATE cxx_std_20)
target_compile_options(pilot_baseline_reentry_reproducer PRIVATE /EHsc /utf-8 /Zc:__cplusplus)
